    18→const TERRAIN_VS = `
    19→attribute vec2 lightv;
    20→attribute vec3 tintv;
    21→varying vec2 vUv;
    22→varying vec2 vLight;
    23→varying vec3 vTint;
    24→varying float vFogDepth;
    25→void main() {
    26→  vUv = uv;
    27→  vLight = lightv;
    28→  vTint = tintv;
    29→  vec4 mv = modelViewMatrix * vec4(position, 1.0);
    30→  vFogDepth = -mv.z;
    31→  gl_Position = projectionMatrix * mv;
    32→}`;
    33→
    34→const TERRAIN_FS = `
    35→uniform sampler2D map;
    36→uniform float dayFactor;
    37→uniform float brightness;
    38→uniform vec3 fogColor;
    39→uniform float fogNear;
    40→uniform float fogFar;
    41→uniform float alpha;
    42→uniform float time;
    43→varying vec2 vUv;
    44→varying vec2 vLight;
    45→varying vec3 vTint;
    46→varying float vFogDepth;
    47→void main() {
    48→  vec2 uv2 = vUv;
    49→  #ifdef WATER
    50→  uv2 += vec2(sin(time * 0.6) * 0.0008, time * 0.0006);
    51→  #endif
    52→  vec4 tex = texture2D(map, uv2);
    53→  #ifdef CUTOUT
    54→  if (tex.a < 0.5) discard;
    55→  #endif
    56→  float sky = vLight.x * dayFactor;
    57→  float L = max(sky, vLight.y);
    58→  float lum = pow(L, 1.45) * 0.96 + 0.04;
    59→  lum = clamp(lum + brightness * (1.0 - lum) * 0.5, 0.0, 1.0);
    60→  vec3 col = tex.rgb * vTint * lum;
    61→  float fogF = smoothstep(fogNear, fogFar, vFogDepth);
    62→  col = mix(col, fogColor, fogF);
    63→  #ifdef CUTOUT
    64→  gl_FragColor = vec4(col, tex.a);
    65→  #else
    66→  gl_FragColor = vec4(col, alpha * tex.a);
    67→  #endif
    68→}`;

   211→  private buildMaterials(): void {
   212→    this.atlasCanvas = buildAtlasCanvas();
   213→    this.atlasTex = new THREE.CanvasTexture(this.atlasCanvas);
   214→    this.atlasTex.magFilter = THREE.NearestFilter;
   215→    this.atlasTex.wrapS = this.atlasTex.wrapT = THREE.ClampToEdgeWrapping;

   217→
   218→    // average tile colors for particles
   219→    for (const blk of BLOCKS) {
   220→      if (!this.tileAvg.has(blk.tiles[2])) {
   221→        this.tileAvg.set(blk.tiles[2], tileAvgColor(this.atlasCanvas, blk.tiles[2]));
   222→      }
   223→    }
   224→
   225→    this.uniforms = {
   226→      map: { value: this.atlasTex },
   227→      dayFactor: { value: 1 },
   228→      brightness: { value: 0.5 },
   229→      fogColor: { value: new THREE.Color(0xbcd8ff) },
   230→      fogNear: { value: 40 },
   231→      fogFar: { value: 120 },
   232→      alpha: { value: 1 },
   233→      time: { value: 0 },
   234→    };

   236→      vertexShader: TERRAIN_VS,
   237→      fragmentShader: TERRAIN_FS,
   238→      uniforms: { ...this.uniforms, alpha: { value: alphaVal } },
   239→      defines,
   240→      transparent,
   241→      ...(transparent ? { depthWrite: false } : {}),
   242→    });

   350→  // ---------------- worker protocol ----------------

   352→  private onWorkerMessage(msg: Record<string, unknown>): void {
   353→    const t = msg.t as string;
   354→    if (t === 'ready') return;
   355→    if (t === 'spawn') {
   356→      this.spawnPoint = { x: msg.x as number, y: msg.y as number, z: msg.z as number };
   357→      if (this.needSpawn && this.mode === 'game') {
   358→        this.player.x = msg.x as number;
   359→        this.player.y = msg.y as number;
   360→        this.player.z = msg.z as number;
   361→        this.needSpawn = false;
   362→      }
   363→      return;
   364→    }
   365→    if (t === 'saved') {
   366→      this.persistWorld(msg.edits as Record<string, number[]>);
   367→      return;
   368→    }
   369→    if (t === 'chunk' || t === 'mesh') {
   370→      const cx = msg.cx as number, cz = msg.cz as number;
   371→      const key = chunkKey(cx, cz);
   372→      let view = this.views.get(key);
   373→      if (!view) {
   374→        view = { cx, cz, blocks: null, light: null, biomes: null, meshes: {} };
   375→        this.views.set(key, view);
   376→      }
   377→      this.pending.delete(key);
   378→      if (t === 'chunk') {
   379→        view.blocks = new Uint8Array(msg.blocks as ArrayBuffer);
   380→        view.light = new Uint8Array(msg.light as ArrayBuffer);
   381→        view.biomes = new Uint8Array(msg.biomes as ArrayBuffer);
   382→      } else if (!view.blocks) {
   383→        // mesh for unknown chunk — ignore
   384→        return;
   385→      }
   386→      this.applyGeom(view, 'solid', msg.solid as GeomData | null);
   387→      this.applyGeom(view, 'cutout', msg.cutout as GeomData | null);
   388→      this.applyGeom(view, 'trans', msg.trans as GeomData | null);
   389→      return;
   390→    }
   391→  }
   392→
   393→  private applyGeom(view: ChunkView, bucket: 'solid' | 'cutout' | 'trans', data: GeomData | null): void {
   394→    const existing = view.meshes[bucket];
   395→    if (!data) {
   396→      if (existing) {
   397→        this.scene.remove(existing);
   398→        existing.geometry.dispose();
   399→        delete view.meshes[bucket];
   400→      }
   401→      return;
   402→    }
   403→    const geo = new THREE.BufferGeometry();
   404→    geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(data.pos), 3));
   405→    geo.setAttribute('uv', new THREE.BufferAttribute(new Float32Array(data.uv), 2));
   406→    geo.setAttribute('lightv', new THREE.BufferAttribute(new Float32Array(data.light), 2));
   407→    geo.setAttribute('tintv', new THREE.BufferAttribute(new Float32Array(data.tint), 3));
   408→    geo.setIndex(new THREE.BufferAttribute(new Uint32Array(data.idx), 1));
   409→    geo.boundingSphere = new THREE.Sphere(new THREE.Vector3(8, WORLD_HEIGHT / 2, 8), 130);
   410→    const mat = bucket === 'solid' ? this.matSolid : bucket === 'cutout' ? this.matCutout : this.matWater;
   411→    if (existing) {
   412→      existing.geometry.dispose();
   413→      existing.geometry = geo;
   414→    } else {
   415→      const mesh = new THREE.Mesh(geo, mat);
   416→      mesh.position.set(view.cx * CHUNK, 0, view.cz * CHUNK);
   417→      mesh.renderOrder = bucket === 'trans' ? 5 : 0;
   418→      mesh.matrixAutoUpdate = false;
   419→      mesh.updateMatrix();
   420→      this.scene.add(mesh);
   421→      view.meshes[bucket] = mesh;
   422→    }
   423→  }
   424→
   425→  private disposeView(v: ChunkView): void {
   426→    for (const k of ['solid', 'cutout', 'trans'] as const) {
   427→      const m = v.meshes[k];
   428→      if (m) {
   429→        this.scene.remove(m);
   430→        m.geometry.dispose();
   431→      }
   432→    }
   433→    this.worker.postMessage({ t: 'unload', cx: v.cx, cz: v.cz });
   434→  }

   438→  private updateStreaming(px: number, pz: number): void {
   439→    const ccx = Math.floor(px / CHUNK), ccz = Math.floor(pz / CHUNK);
   440→    if (ccx === this.lastChunkX && ccz === this.lastChunkZ && this.pending.size < 160) return;
   441→    this.lastChunkX = ccx; this.lastChunkZ = ccz;
   442→    const rd = this.wantedRenderDist;
   443→    const wanted: { key: number; cx: number; cz: number; d: number }[] = [];
   444→    for (let dz = -rd; dz <= rd; dz++)
   445→      for (let dx = -rd; dx <= rd; dx++) {
   446→        const d = dx * dx + dz * dz;
   447→        if (d > (rd + 0.5) * (rd + 0.5)) continue;
   448→        const cx = ccx + dx, cz = ccz + dz;
   449→        const key = chunkKey(cx, cz);
   450→        if (this.views.has(key) || this.pending.has(key)) continue;
   451→        wanted.push({ key, cx, cz, d });
   452→      }
   453→    wanted.sort((a, b) => a.d - b.d);
   454→    for (const w of wanted) {
   455→      if (this.pending.size >= 220) break;
   456→      this.pending.add(w.key);
   457→      this.worker.postMessage({ t: 'gen', cx: w.cx, cz: w.cz });
   458→    }
   459→    // unload far chunks
   460→    const maxD = (rd + 3) * (rd + 3);
   461→    for (const [key, v] of this.views) {
   462→      const dx = v.cx - ccx, dz = v.cz - ccz;
   463→      if (dx * dx + dz * dz > maxD) {
   464→        this.disposeView(v);
   465→        this.views.delete(key);
   466→      }
   467→    }
   468→  }

   470→  // ---------------- world access ----------------
   471→
   472→  getBlock(x: number, y: number, z: number): number {
   473→    if (y < 0) return 17; // bedrock
   474→    if (y >= WORLD_HEIGHT) return AIR;
   475→    const v = this.views.get(chunkKey(x >> 4, z >> 4));
   476→    if (!v || !v.blocks) return 1; // unloaded treated as stone (solid barrier)
   477→    return v.blocks[(y << 8) | ((z & 15) << 4) | (x & 15)];
   478→  }
   479→
   480→  chunkLoaded(x: number, z: number): boolean {
   481→    const v = this.views.get(chunkKey(x >> 4, z >> 4));
   482→    return !!(v && v.blocks);
   483→  }
   484→
   485→  setBlock(x: number, y: number, z: number, id: number): void {
   486→    if (y < 1 || y >= WORLD_HEIGHT) return;
   487→    const v = this.views.get(chunkKey(x >> 4, z >> 4));
   488→    if (v && v.blocks) v.blocks[(y << 8) | ((z & 15) << 4) | (x & 15)] = id;
   489→    this.worker.postMessage({ t: 'edit', x, y, z, id });
   490→  }

   565→  private handleKey(e: KeyboardEvent, down: boolean): void {
   566→    const ui = useUI.getState();
   567→    const s = useSettings.getState().settings;
   568→    const code = e.code;
   569→
   570→    // always track raw keys so nothing gets stuck
   571→    if (down) this.keysDown.add(code);
   572→    else this.keysDown.delete(code);
   573→
   574→    // global function keys
   575→    if (down && code === s.keys.toggleDebug && (ui.screen === 'playing' || ui.screen === 'paused')) {
   576→      e.preventDefault();
   577→      ui.set({ debugOpen: !ui.debugOpen });
   578→      return;
   579→    }
   580→    if (down && code === s.keys.screenshot && ui.screen === 'playing') {
   581→      e.preventDefault();
   582→      void this.screenshot();
   583→      return;
   584→    }
   585→    if (down && code === s.keys.toggleFullscreen && ui.screen !== 'options') {
   586→      e.preventDefault();
   587→      useSettings.getState().update('fullscreen', !s.fullscreen);
   588→      return;
   589→    }
   590→    if (down && code === 'Escape') {
   591→      if (ui.screen === 'paused') {
   592→        ui.set({ screen: 'playing', pauseReason: null });
   593→        this.resumeLock();
   594→        return;
   595→      }
   596→    }
   597→    if (ui.screen !== 'playing') return;
   598→
   599→    if (down && code === s.keys.inventory) {
   600→      e.preventDefault();
   601→      ui.set({ screen: 'inventory' });
   602→      document.exitPointerLock();
   603→      this.sound.click();
   604→      return;
   605→    }
   606→    if (down && code.startsWith('Digit')) {
   607→      const n = parseInt(code.slice(5), 10);
   608→      if (n >= 1 && n <= 9) ui.set({ selectedSlot: n - 1 });
   609→    }
   610→    if (down && code === s.keys.jump) {
   611→      const now = performance.now();
   612→      if (now - this.lastSpaceTap < 280 && this.player.mode === 'creative') {
   613→        this.player.flying = !this.player.flying;
   614→        this.player.vy = 0;
   615→        this.lastSpaceTap = 0;
   616→      } else {
   617→        this.lastSpaceTap = now;
   618→      }
   619→    }
   620→    if (down && code === s.keys.forward) {
   621→      const now = performance.now();
   622→      if (now - this.lastWTap < 280) this.sprintByDouble = true;
   623→      this.lastWTap = now;
   624→    }
   625→    if (down && code === 'F5') e.preventDefault();
   626→  }

   664→  private raycast(): { x: number; y: number; z: number; fx: number; fy: number; fz: number } | null {
   665→    const eye = { x: 0, y: 0, z: 0 };
   666→    this.player.eyePos(eye);
   667→    const dir = new THREE.Vector3(0, 0, -1).applyEuler(new THREE.Euler(this.player.pitch, this.player.yaw, 0, 'YXZ'));
   668→    let x = Math.floor(eye.x), y = Math.floor(eye.y), z = Math.floor(eye.z);
   669→    const stepX = dir.x > 0 ? 1 : -1, stepY = dir.y > 0 ? 1 : -1, stepZ = dir.z > 0 ? 1 : -1;
   670→    const tDeltaX = Math.abs(1 / (dir.x || 1e-9)), tDeltaY = Math.abs(1 / (dir.y || 1e-9)), tDeltaZ = Math.abs(1 / (dir.z || 1e-9));
   671→    let tMaxX = tDeltaX * (dir.x > 0 ? (x + 1 - eye.x) : (eye.x - x));
   672→    let tMaxY = tDeltaY * (dir.y > 0 ? (y + 1 - eye.y) : (eye.y - y));
   673→    let tMaxZ = tDeltaZ * (dir.z > 0 ? (z + 1 - eye.z) : (eye.z - z));
   674→    let fx = 0, fy = 0, fz = 0;
   675→    const maxDist = 4.5;
   676→    for (let i = 0; i < 128; i++) {
   677→      const id = this.getBlock(x, y, z);
   678→      if (id !== AIR && id !== WATER) {
   679→        return { x, y, z, fx, fy, fz };
   680→      }
   681→      if (tMaxX < tMaxY && tMaxX < tMaxZ) {
   682→        if (tMaxX > maxDist) break;
   683→        x += stepX; tMaxX += tDeltaX; fx = -stepX; fy = 0; fz = 0;
   684→      } else if (tMaxY < tMaxZ) {
   685→        if (tMaxY > maxDist) break;
   686→        y += stepY; tMaxY += tDeltaY; fx = 0; fy = -stepY; fz = 0;
   687→      } else {
   688→        if (tMaxZ > maxDist) break;
   689→        z += stepZ; tMaxZ += tDeltaZ; fx = 0; fy = 0; fz = -stepZ;
   690→      }
   691→    }
   692→    return null;
   693→  }

   695→  private updateInteraction(dt: number): void {
   696→    const ui = useUI.getState();
   697→    if (this.mode !== 'game' || this.paused || this.player.dead) {
   698→      this.outline.visible = false;
   699→      this.crackMesh.visible = false;
   700→      return;
   701→    }
   702→    const hit = this.raycast();
   703→    this.currentHit = hit;
   704→    if (hit) {
   705→      this.outline.visible = true;
   706→      this.outline.position.set(hit.x + 0.5, hit.y + 0.5, hit.z + 0.5);
   707→    } else {
   708→      this.outline.visible = false;
   709→    }
   710→
   711→    const s = useSettings.getState().settings;
   712→    const creative = this.player.mode === 'creative';
   713→    const selectedId = ui.hotbar[ui.selectedSlot] ?? 0;
   714→
   715→    // breaking
   716→    if (this.mouseL && hit) {
   717→      const changed = !this.breakTarget || this.breakTarget.x !== hit.x || this.breakTarget.y !== hit.y || this.breakTarget.z !== hit.z;
   718→      if (changed) {
   719→        this.breakTarget = { x: hit.x, y: hit.y, z: hit.z };
   720→        this.breakProgress = 0;
   721→      }
   722→      const id = this.getBlock(hit.x, hit.y, hit.z);
   723→      const hard = creative ? 0.06 : Math.max(0.05, HARDNESS[id] || 0.5);
   724→      this.breakProgress += dt / hard;
   725→      this.digSoundTimer -= dt;
   726→      if (this.digSoundTimer <= 0) {
   727→        this.digSoundTimer = 0.22;
   728→        this.sound.dig(SOUND_OF[id] as never, 0.9 + Math.random() * 0.2);
   729→      }
   730→      if (this.breakProgress >= 1) {
   731→        this.destroyBlock(hit.x, hit.y, hit.z);
   732→        this.breakProgress = 0;
   733→        this.breakTarget = null;
   734→        this.creativeBreakTimer = 0.25;
   735→      }
   736→    } else {
   737→      this.breakTarget = null;
   738→      this.breakProgress = 0;
   739→    }
   740→
   741→    // crack overlay
   742→    if (this.breakTarget && this.breakProgress > 0.02 && !creative) {
   743→      const stage = Math.min(9, Math.floor(this.breakProgress * 10));
   744→      this.crackMesh.visible = true;
   745→      this.crackMesh.position.set(this.breakTarget.x + 0.5, this.breakTarget.y + 0.5, this.breakTarget.z + 0.5);
   746→      const mat = this.crackMesh.material as THREE.MeshBasicMaterial;
   747→      if (mat.map !== this.crackTex[stage]) {
   748→        mat.map = this.crackTex[stage];
   749→        mat.needsUpdate = true;
   750→      }
   751→    } else {
   752→      this.crackMesh.visible = false;
   753→    }
   754→
   755→    // placing
   756→    this.placeCooldown -= dt;
   757→    this.creativeBreakTimer -= dt;
   758→    if (this.mouseR && hit && this.placeCooldown <= 0) {
   759→      this.placeCooldown = 0.22;
   760→      const px2 = hit.x + hit.fx, py2 = hit.y + hit.fy, pz2 = hit.z + hit.fz;
   761→      if (py2 >= 1 && py2 < WORLD_HEIGHT) {
   762→        const existing = this.getBlock(px2, py2, pz2);
   763→        const replaceable = existing === AIR || existing === WATER || CROSS[existing] === 1;
   764→        if (replaceable && selectedId !== 0) {
   765→          // player AABB overlap check for solid blocks
   766→          const solid = IS_SOLID[selectedId] === 1;
   767→          let blocked = false;
   768→          if (solid) {
   769→            const p = this.player;
   770→            const px0 = p.x - 0.31, px1 = p.x + 0.31, pz0 = p.z - 0.31, pz1 = p.z + 0.31;
   771→            const py0 = p.y, py1 = p.y + 1.8;
   772→            blocked = px1 > px2 && px0 < px2 + 1 && py1 > py2 && py0 < py2 + 1 && pz1 > pz2 && pz0 < pz2 + 1;
   773→          }
   774→          if (!blocked) {
   775→            this.setBlock(px2, py2, pz2, selectedId);
   776→            this.sound.place(SOUND_OF[selectedId] as never);
   777→          }
   778→        }
   779→      }
   780→    }
   781→  }

   783→  private destroyBlock(x: number, y: number, z: number): void {
   784→    const id = this.getBlock(x, y, z);
   785→    if (id === AIR) return;

   792→    this.particles.spawnBlockBreak(x, y, z, color, count);
   793→  }

   842→  private scheduleFrame(): void {
   843→    if (!this.running || this.disposed) return;
   844→    const s = useSettings.getState().settings;
   845→    if (s.vsync) {
   846→      this.rafId = requestAnimationFrame(() => this.loop());
   847→    } else {
   848→      this.timeoutId = window.setTimeout(() => this.loop(), 0);
   849→    }
   850→  }
   851→
   852→  private loop(): void {
   853→    if (!this.running || this.disposed) return;
   854→    const now = performance.now();
   855→    let dt = (now - this.lastFrame) / 1000;
   856→    this.lastFrame = now;
   857→    if (dt > 0.25) dt = 0.25;
   858→    if (dt < 0) dt = 0;
   859→    this.fpsEMA = this.fpsEMA * 0.95 + (1 / Math.max(dt, 1e-4)) * 0.05;
   860→
   861→    // fps cap
   862→    const s = useSettings.getState().settings;
   863→    const interval = s.maxFps > 0 ? 1000 / s.maxFps : 0;
   864→    if (interval > 0 && now - this.frameAcc < interval - 0.5) {
   865→      this.scheduleFrame();
   866→      return;
   867→    }
   868→    this.frameAcc = now;
   869→
   870→    this.update(dt);
   871→    this.renderFrame();
   872→    this.renderCalls++;
   873→    this.scheduleFrame();
   874→  }

   876→  private update(dt: number): void {
   877→    const s = useSettings.getState().settings;
   878→    const ui = useUI.getState();
   879→    this.uniforms.time.value = performance.now() / 1000;
   880→
   881→    if (this.mode === 'panorama') {
   882→      this.panoramaAngle += dt * 0.05;
   883→      const cx = 8, cz = 8;
   884→      const r = 34;
   885→      this.camera.position.set(cx + Math.cos(this.panoramaAngle) * r, 88, cz + Math.sin(this.panoramaAngle) * r);
   886→      this.camera.lookAt(cx + Math.cos(this.panoramaAngle + 2.4) * 10, 74, cz + Math.sin(this.panoramaAngle + 2.4) * 10);
   887→      this.camera.fov = 62;
   888→      this.camera.updateProjectionMatrix();
   889→      this.wantedRenderDist = Math.min(s.renderDistance, 6);
   890→      this.updateStreaming(cx, cz);
   891→      this.sky.update(dt, this.camera.position.x, this.camera.position.y, this.camera.position.z, false);
   892→      this.updateUniformsCommon(dt, false);
   893→      this.particles.update(dt);
   894→      return;
   895→    }
   896→    if (this.mode !== 'game') return;
   897→
   898→    // input state
   899→    const k = s.keys;
   900→    const kd = this.keysDown;
   901→    this.input.fwd = kd.has(k.forward);
   902→    this.input.back = kd.has(k.back);
   903→    this.input.left = kd.has(k.left);
   904→    this.input.right = kd.has(k.right);
   905→    this.input.jump = kd.has(k.jump);
   906→    this.input.sneak = kd.has(k.sneak);
   907→    const wantSprint = kd.has(k.sprint) || this.sprintByDouble;
   908→    this.input.sprint = wantSprint;
   909→    if (!this.input.fwd) this.sprintByDouble = false;
   910→
   911→    // freeze physics while the loading screen is up
   912→    const frozen = this.paused || this.player.dead || ui.screen === 'loading';
   913→    if (!frozen) {
   914→      // fixed substeps for physics
   915→      let remaining = dt;
   916→      while (remaining > 0) {
   917→        const step = Math.min(remaining, 1 / 60);
   918→        this.player.update(step, this.input, this);
   919→        remaining -= step;
   920→      }
   921→    }
   922→
   923→    // streaming
   924→    this.updateStreaming(this.player.x, this.player.z);
   925→
   926→    // camera
   927→    this.updateCamera(dt, s);
   928→
   929→    // sky + fog
   930→    this.underwater = this.player.headInWater;
   931→    this.sky.update(dt, this.camera.position.x, this.camera.position.y, this.camera.position.z, this.underwater);
   932→    this.updateUniformsCommon(dt, this.underwater);
   933→
   934→    // interaction
   935→    this.updateInteraction(dt);
   936→    this.particles.update(dt);
   937→
   938→    // cave ambience
   939→    this.caveDroneTimer -= dt;
   940→    if (this.caveDroneTimer <= 0) {
   941→      this.caveDroneTimer = 25 + Math.random() * 40;
   942→      const v = this.views.get(chunkKey(Math.floor(this.player.x) >> 4, Math.floor(this.player.z) >> 4));
   943→      if (v && v.light) {
   944→        const li = v.light[((Math.floor(this.player.y) << 8) | ((Math.floor(this.player.z) & 15) << 4) | (Math.floor(this.player.x) & 15))];
   945→        if ((li >> 4) < 4 && this.player.y < 55) this.sound.caveDrone();
   946→      }
   947→    }
   948→
   949→    // landing damage hookup
   950→    if (this.player.onLand === null) {
   951→      this.player.onLand = (fall) => {
   952→        this.player.landDamage(fall);
   953→      };
   954→      this.player.onDamage = () => {
   955→        this.sound.hurt();
   956→        useUI.getState().set({ hitFlash: performance.now() });
   957→      };
   958→      this.player.onStep = (fam) => this.sound.step(fam as never);
   959→      this.player.onSplash = () => this.sound.splash();
   960→      this.player.onDeath = () => {
   961→        useUI.getState().set({ screen: 'dead', deathMessage: 'You fell out of the world' });
   962→        document.exitPointerLock();
   963→      };
   964→    }
   965→
   966→    // autosave
   967→    this.saveTimer += dt;
   968→    if (this.saveTimer > 20) {
   969→      this.saveTimer = 0;
   970→      this.saveNow();
   971→    }
   972→
   973→    // debug info
   974→    const p = this.player;
   975→    this.debug.fps = Math.round(this.fpsEMA);
   976→    this.debug.x = p.x; this.debug.y = p.y; this.debug.z = p.z;
   977→    this.debug.chunkX = Math.floor(p.x / 16); this.debug.chunkZ = Math.floor(p.z / 16);
   978→    const yawDeg = ((-p.yaw * 180 / Math.PI) % 360 + 360) % 360;
   979→    const dirs = ['south', 'west', 'north', 'east'];
   980→    this.debug.facing = dirs[Math.round(yawDeg / 90) % 4];
   981→    this.debug.yaw = yawDeg;
   982→    this.debug.pitch = p.pitch * 180 / Math.PI;
   983→    const cv = this.views.get(chunkKey(Math.floor(p.x) >> 4, Math.floor(p.z) >> 4));
   984→    if (cv && cv.biomes) {
   985→      this.debug.biome = BIOME_NAMES[cv.biomes[(Math.floor(p.z) & 15) * 16 + (Math.floor(p.x) & 15)]] || '—';
   986→    }
   987→    if (cv && cv.light) {
   988→      this.debug.lightFeet = cv.light[((Math.floor(p.y) << 8) | ((Math.floor(p.z) & 15) << 4) | (Math.floor(p.x) & 15))] >> 4;
   989→    }

  1016→  private updateCamera(dt: number, s: Settings): void {
  1017→    const p = this.player;
  1018→    // fov: sprint kick
  1019→    const hSpeed = Math.hypot(p.vx, p.vz);
  1020→    const sprinting = p.sprinting || (hSpeed > 4.8);
  1021→    const targetFov = s.fov * (sprinting && hSpeed > 1 ? 1.12 : 1) * (p.flying && hSpeed > 12 ? 1.06 : 1);
  1022→    this.camera.fov += (targetFov - this.camera.fov) * Math.min(1, dt * 8);
  1023→    this.camera.updateProjectionMatrix();
  1024→
  1025→    const eyeY = p.y + p.eyeHeight;
  1026→    this.camera.rotation.order = 'YXZ';
  1027→    this.camera.rotation.set(p.pitch, p.yaw, 0);
  1028→
  1029→    let bx = 0, by = 0, roll = 0;
  1030→    if (s.viewBobbing && p.bobAmp > 0.01) {
  1031→      const amp = p.bobAmp * Math.min(1, hSpeed / 4.3);
  1032→      bx = Math.cos(p.bobPhase) * 0.02 * amp;
  1033→      by = Math.abs(Math.sin(p.bobPhase)) * 0.055 * amp;
  1034→      roll = Math.cos(p.bobPhase) * 0.0035 * amp;
  1035→    }
  1036→    // camera position relative to look direction for bob
  1037→    const cos = Math.cos(p.yaw), sin = Math.sin(p.yaw);
  1038→    this.camera.position.set(p.x + bx * cos, eyeY + by, p.z - bx * sin);
  1039→    this.camera.rotation.z = roll;
  1040→  }

  1103→  /** 0..1 progress of the initial ring of chunks around the player */
  1104→  loadingProgress(): number {
  1105→    if (this.mode !== 'game') return 1;
  1106→    const r = 3;
  1107→    const ccx = Math.floor(this.player.x / CHUNK), ccz = Math.floor(this.player.z / CHUNK);
  1108→    let need = 0, have = 0;
  1109→    for (let dz = -r; dz <= r; dz++)
  1110→      for (let dx = -r; dx <= r; dx++) {
  1111→        if (dx * dx + dz * dz > (r + 0.5) * (r + 0.5)) continue;
  1112→        need++;
  1113→        const v = this.views.get(chunkKey(ccx + dx, ccz + dz));
  1114→        if (v && v.blocks) have++;
  1115→      }
  1116→    return need === 0 ? 1 : have / need;
  1117→  }
